Chat 이미지·파일 첨부 처리 Implementation Plan ​

For agentic workers: REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (- [ ]) syntax for tracking.

Goal: main-site aicoding 채팅의 이미지(인라인 image block)·파일(cwd 하위 저장 후 경로 전달) 첨부를 node-executor/executor.mjs 에서 처리한다.

Architecture: 릴레이→데몬이 브라우저 JSON 을 executor stdin 에 verbatim 전달하므로 변경은 executor.mjs 한 파일뿐이다. req.images 는 user 메시지에 image content block 으로 인라인, req.files 는 <cwd>/.kroot/uploads/<turnId>/ 에 저장하고 절대경로를 프롬프트에 append 한다. 순수 함수(검증·파일 처리)는 node --test 로 단위 검증한다.

Tech Stack: Node ≥ 20 (ESM, node:fs sync API, node:test), @anthropic-ai/claude-agent-sdk.

Spec: docs/superpowers/specs/2026-07-06-chat-image-file-attachments-design.md

// or the kroot-chat-executor bin) — not when imported by tests, where main()
// would block forever on stdin. realpath-normalize both sides: symlinked bins

  isMainModule = realpathSync(process.argv[1]) === fileURLToPath(import.meta.url)

  // argv[1] missing/unresolvable — not run as a script.
}
if (isMainModule) {
  main().catch((e) => {
    out({ type: 'error', message: String(e?.message ?? e) })
    process.exit(1)
  })
}

export const userMessage = (text, images) => {
  const content = []
  if (Array.isArray(images)) {
    for (const img of images) {
      content.push({ type: 'image', source: { type: 'base64', media_type: img.mimeType, data: img.data } })
    }
  }
  content.push({ type: 'text', text })
  return { type: 'user', message: { role: 'user', content }, parent_tool_use_id: null }
}

// Attachment bounds. These are a BACKSTOP: the relay and daemon both cap a ws
// frame at 16 MiB, so the real ceiling is the total frame. See the spec.
const MAX_IMAGE_COUNT = 20
const MAX_FILE_COUNT = 20
const MAX_TOTAL_ATTACHMENT_CHARS = 12 * 1024 * 1024 // base64 chars, under the 16 MiB frame limit
const SUPPORTED_IMAGE_MIME_TYPES = new Set(['image/jpeg', 'image/png', 'image/gif', 'image/webp'])

export function validateImages(images) {
  if (images === undefined) return true
  return (
    Array.isArray(images) &&
    images.length <= MAX_IMAGE_COUNT &&
    images.every(
      (img) =>
        img &&
        typeof img.data === 'string' &&
        img.data.length > 0 &&
        typeof img.mimeType === 'string' &&
        SUPPORTED_IMAGE_MIME_TYPES.has(img.mimeType),
    )
  )
}

export function validateFiles(files) {
  if (files === undefined) return true
  return (
    Array.isArray(files) &&
    files.length <= MAX_FILE_COUNT &&
    files.every(
      (f) =>
        f &&
        typeof f.name === 'string' &&
        f.name.length > 0 &&
        typeof f.data === 'string' &&
        f.data.length > 0 &&
        (f.mimeType === undefined || typeof f.mimeType === 'string'),
    )
  )
}

export function withinAttachmentBudget(images, files) {
  let total = 0
  if (Array.isArray(images)) for (const img of images) total += (img?.data?.length) || 0
  if (Array.isArray(files)) for (const f of files) total += (f?.data?.length) || 0
  return total <= MAX_TOTAL_ATTACHMENT_CHARS
}

export function sanitizeFilename(name) {
  // basename strips POSIX directory components (path traversal defense);
  // then drop control chars and any remaining separators (Windows-style \).
  let base = path.basename(String(name)).replace(/[\x00-\x1f]/g, '').replace(/[/\\]/g, '_').trim()
  if (base === '' || base === '.' || base === '..') base = 'file'
  return base
}

// materializeFiles writes each uploaded file (base64) into a fresh per-turn
// folder under <cwd>/.kroot/uploads/, sanitizing names and renaming collisions.
// Returns absolute paths and a note listing them (appended to the user prompt so
// the agent knows the files exist and where). Throws on I/O failure.
export function materializeFiles(cwd, files) {
  if (!Array.isArray(files) || files.length === 0) return { paths: [], note: '' }
  const turnId = `${Date.now()}-${randomUUID().slice(0, 8)}`
  const dir = path.join(cwd, '.kroot', 'uploads', turnId)
  mkdirSync(dir, { recursive: true })
  const used = new Set()
  const paths = []
  for (const f of files) {
    let name = sanitizeFilename(f.name)
    if (used.has(name)) {
      const ext = path.extname(name)
      const stem = name.slice(0, name.length - ext.length)
      let i = 2
      while (used.has(`${stem} (${i})${ext}`)) i++
      name = `${stem} (${i})${ext}`
    }
    used.add(name)
    const abs = path.join(dir, name)
    writeFileSync(abs, Buffer.from(f.data, 'base64'))
    paths.push(abs)
  }
  const note = '\n\n[첨부된 파일 — 로컬 경로]\n' + paths.map((p) => `- ${p}`).join('\n')
  return { paths, note }
}

function handleChat(query, req) {
  const text = req.prompt || req.text || ''
  if (!text.trim()) {
    out({ type: 'error', message: 'Empty prompt' })
    return
  }
  // Validate attachments up front (bool checks, not throw-in-buildOptions): the
  // live-session fast path below never calls buildOptions, so a validator there
  // would silently never run on that path. Fail closed with a clear error.
  if (!validateImages(req.images)) {
    out({ type: 'error', message: 'invalid images: array of {data:string, mimeType:string} with a supported mimeType' })
    return
  }
  if (!validateFiles(req.files)) {
    out({ type: 'error', message: 'invalid files: array of {name:string, data:string} objects' })
    return
  }
  if (!withinAttachmentBudget(req.images, req.files)) {
    out({ type: 'error', message: 'attachments too large: total base64 exceeds the 12 MiB frame budget' })
    return
  }
  const cwd = resolveCwd(req)

  // Save uploaded files to disk and append their paths to the prompt so the
  // agent can Read/edit them with its file tools.
  let messageText = text
  try {
    const { note } = materializeFiles(cwd, req.files)
    if (note) messageText += note
  } catch (e) {
    out({ type: 'error', message: `failed to save attached files: ${String(e?.message ?? e)}` })
    return
  }

  if (active && !active.closed && active.cwd === cwd && req.sessionId &&
      (req.sessionId === active.capturedSessionId || req.sessionId === active.requestedSessionId)) {
    active.openTurns++
    if (active.input.push(userMessage(messageText, req.images))) { debug('chat → pushed into live query'); return }
    active.openTurns--
    // input already ended → fall through to a fresh query
  }

  if (active && !active.closed) active.input.end()

  const input = makeInputStream()
  input.push(userMessage(messageText, req.images))

## 첨부 (이미지·파일)

chat 요청 JSON 에 선택적으로 첨부를 넣는다. main-site 는 순수 base64(데이터 URI 접두부 제거)로
