kroot-adk Chat — 릴레이 서버 (Option D2, Step 2+3) Implementation Plan ​

For agentic workers: REQUIRED SUB-SKILL: superpowers:subagent-driven-development. Steps use checkbox (- [ ]).

Goal: kroot-adk 안에 Go ws 릴레이 서버(kroot relay serve)를 만든다. 브라우저 leg(/ws/browser)와 데몬 leg(/ws/agent)를 user 단위로 묶어 메시지를 브리지한다. 라우팅/매칭 코어(Registry)는 ws 없이 단위 테스트하고, 그 위에 coder/websocket 전송을 얹는다.

  • 릴레이 = kroot-adk 내 Go 서비스(클라이언트/데몬과 같은 레포·언어 → 함께 테스트). 배포: 서버에서 kroot relay serve(뒤에 Traefik wss).
  • 양다리 ws↔ws(단일 프로토콜 브리지). 메시지는 executor JSON 재사용.
  • 인증은 Step 5(PAT) 로 분리 — Step 3 은 dev 식별자(?user=, ?token=)로 매칭만.

Tech: Go 1.25 + github.com/coder/websocket. 레포 github.com/kslaboratory/kroot-adk, 브랜치 feat/chat-connect-claude-driver.

import (
	"sync"
	"testing"
)

type fakeSender struct {
	mu   sync.Mutex
	got  [][]byte
}

func (f *fakeSender) Send(m []byte) error {
	f.mu.Lock()
	defer f.mu.Unlock()
	f.got = append(f.got, append([]byte(nil), m...))
	return nil
}
func (f *fakeSender) count() int {
	f.mu.Lock()
	defer f.mu.Unlock()
	return len(f.got)
}

func TestRegistry(t *testing.T) {
	r := NewRegistry()

	// no agent yet → browser bind fails
	br := &fakeSender{}
	if _, ok, _ := r.RegisterBrowser("u", br); ok {
		t.Fatal("browser bound with no agent")
	}

	// register agent, then browser binds
	ag := &fakeSender{}
	agUnreg := r.RegisterAgent("u", ag)
	agent, ok, brUnreg := r.RegisterBrowser("u", br)
	if !ok || agent == nil {
		t.Fatal("browser should bind to agent")
	}

	// routing lookups
	if s, ok := r.AgentFor("u"); !ok || s != ag {
		t.Errorf("AgentFor(u) mismatch")
	}
	if s, ok := r.BrowserFor("u"); !ok || s != br {
		t.Errorf("BrowserFor(u) mismatch")
	}
	if _, ok := r.AgentFor("other"); ok {
		t.Errorf("AgentFor(other) should be false")
	}

	// unregister browser → BrowserFor gone, agent stays
	brUnreg()
	if _, ok := r.BrowserFor("u"); ok {
		t.Errorf("BrowserFor(u) should be gone after unregister")
	}
	if _, ok := r.AgentFor("u"); !ok {
		t.Errorf("AgentFor(u) should remain")
	}

	// unregister agent → gone
	agUnreg()
	if _, ok := r.AgentFor("u"); ok {
		t.Errorf("AgentFor(u) should be gone")
	}
}

go
// Package relay bridges a browser websocket and its user's agent websocket.
package relay

import "sync"

// Sender sends one text ws message to a peer.
type Sender interface {
	Send(msg []byte) error
}

// Registry matches, per user, a browser Sender to that user's agent Sender.
// MVP: one agent and one browser per user (1:1).
type Registry struct {
	mu       sync.Mutex
	agents   map[string]Sender
	browsers map[string]Sender
}

// NewRegistry returns an empty Registry.
func NewRegistry() *Registry {
	return &Registry{agents: map[string]Sender{}, browsers: map[string]Sender{}}
}

// RegisterAgent registers user's agent and returns a func to unregister it.
func (r *Registry) RegisterAgent(user string, s Sender) (unregister func()) {
	r.mu.Lock()
	r.agents[user] = s
	r.mu.Unlock()
	return func() {
		r.mu.Lock()
		if r.agents[user] == s {
			delete(r.agents, user)
		}
		r.mu.Unlock()
	}
}

// RegisterBrowser binds a browser to user's agent. ok=false if no agent is
// connected for that user. Returns the agent Sender and an unregister func.
func (r *Registry) RegisterBrowser(user string, s Sender) (agent Sender, ok bool, unregister func()) {
	r.mu.Lock()
	defer r.mu.Unlock()
	a, exists := r.agents[user]
	if !exists {
		return nil, false, func() {}
	}
	r.browsers[user] = s
	return a, true, func() {
		r.mu.Lock()
		if r.browsers[user] == s {
			delete(r.browsers, user)
		}
		r.mu.Unlock()
	}
}

// AgentFor returns the agent Sender for user.
func (r *Registry) AgentFor(user string) (Sender, bool) {
	r.mu.Lock()
	defer r.mu.Unlock()
	s, ok := r.agents[user]
	return s, ok
}

// BrowserFor returns the browser Sender for user.
func (r *Registry) BrowserFor(user string) (Sender, bool) {
	r.mu.Lock()
	defer r.mu.Unlock()
	s, ok := r.browsers[user]
	return s, ok
}

package relay

import (
	"context"
	"net/http/httptest"
	"strings"
	"testing"
	"time"

	"github.com/coder/websocket"
)

func TestServer_Bridge(t *testing.T) {
	srv := httptest.NewServer(NewServer())
	defer srv.Close()
	base := "ws" + strings.TrimPrefix(srv.URL, "http")

	ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
	defer cancel()

	// agent connects first
	agent, _, err := websocket.Dial(ctx, base+"/ws/agent?user=u", nil)
	if err != nil {
		t.Fatalf("agent dial: %v", err)
	}
	defer agent.Close(websocket.StatusNormalClosure, "")

	// give the server a moment to register the agent
	time.Sleep(100 * time.Millisecond)

	// browser connects
	browser, _, err := websocket.Dial(ctx, base+"/ws/browser?user=u", nil)
	if err != nil {
		t.Fatalf("browser dial: %v", err)
	}
	defer browser.Close(websocket.StatusNormalClosure, "")

	// browser → agent
	if err := browser.Write(ctx, websocket.MessageText, []byte(`{"type":"chat","prompt":"hi"}`)); err != nil {
		t.Fatalf("browser write: %v", err)
	}
	_, got, err := agent.Read(ctx)
	if err != nil {
		t.Fatalf("agent read: %v", err)
	}
	if !strings.Contains(string(got), `"prompt":"hi"`) {
		t.Errorf("agent got %q", got)
	}

	// agent → browser
	if err := agent.Write(ctx, websocket.MessageText, []byte(`{"type":"text","text":"pong"}`)); err != nil {
		t.Fatalf("agent write: %v", err)
	}
	_, got2, err := browser.Read(ctx)
	if err != nil {
		t.Fatalf("browser read: %v", err)
	}
	if !strings.Contains(string(got2), `"text":"pong"`) {
		t.Errorf("browser got %q", got2)
	}
}

func TestServer_NoAgent(t *testing.T) {
	srv := httptest.NewServer(NewServer())
	defer srv.Close()
	base := "ws" + strings.TrimPrefix(srv.URL, "http")
	ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
	defer cancel()

	browser, _, err := websocket.Dial(ctx, base+"/ws/browser?user=nobody", nil)
	if err != nil {
		t.Fatalf("browser dial: %v", err)
	}
	defer browser.Close(websocket.StatusNormalClosure, "")
	_, got, err := browser.Read(ctx)
	if err != nil {
		t.Fatalf("browser read: %v", err)
	}
	if !strings.Contains(string(got), "agent:unavailable") {
		t.Errorf("expected agent:unavailable, got %q", got)
	}
}

coder/websocket API 기준: websocket.Accept(w, r, *AcceptOptions) (*Conn, error), c.Read(ctx) (MessageType, []byte, error), c.Write(ctx, MessageType, []byte) error, c.Close(StatusCode, reason). 컴파일 후 시그니처가 다르면 lib(go doc github.com/coder/websocket)로 확인해 맞추세요.

go
package relay

import (
	"context"
	"net/http"
	"sync"

	"github.com/coder/websocket"
)

// Server is the ws relay HTTP handler.
type Server struct {
	reg *Registry
	mux *http.ServeMux
}

// NewServer returns a relay Server (http.Handler).
func NewServer() *Server {
	s := &Server{reg: NewRegistry(), mux: http.NewServeMux()}
	s.mux.HandleFunc("/ws/agent", s.handleAgent)
	s.mux.HandleFunc("/ws/browser", s.handleBrowser)
	return s
}

func (s *Server) ServeHTTP(w http.ResponseWriter, r *http.Request) { s.mux.ServeHTTP(w, r) }

// wsSender adapts a websocket.Conn to Sender (serialized writes).
type wsSender struct {
	ctx context.Context
	c   *websocket.Conn
	mu  sync.Mutex
}

func (s *wsSender) Send(msg []byte) error {
	s.mu.Lock()
	defer s.mu.Unlock()
	return s.c.Write(s.ctx, websocket.MessageText, msg)
}

func (s *Server) handleAgent(w http.ResponseWriter, r *http.Request) {
	user := r.URL.Query().Get("user")
	if user == "" {
		http.Error(w, "missing user", http.StatusBadRequest)
		return
	}
	c, err := websocket.Accept(w, r, &websocket.AcceptOptions{InsecureSkipVerify: true})
	if err != nil {
		return
	}
	defer c.Close(websocket.StatusNormalClosure, "")
	ctx := r.Context()
	unreg := s.reg.RegisterAgent(user, &wsSender{ctx: ctx, c: c})
	defer unreg()
	for {
		_, data, err := c.Read(ctx)
		if err != nil {
			return
		}
		if b, ok := s.reg.BrowserFor(user); ok {
			_ = b.Send(data)
		}
	}
}

func (s *Server) handleBrowser(w http.ResponseWriter, r *http.Request) {
	user := r.URL.Query().Get("user")
	if user == "" {
		http.Error(w, "missing user", http.StatusBadRequest)
		return
	}
	c, err := websocket.Accept(w, r, &websocket.AcceptOptions{InsecureSkipVerify: true})
	if err != nil {
		return
	}
	defer c.Close(websocket.StatusNormalClosure, "")
	ctx := r.Context()
	sender := &wsSender{ctx: ctx, c: c}
	_, ok, unreg := s.reg.RegisterBrowser(user, sender)
	if !ok {
		_ = sender.Send([]byte(`{"type":"agent:unavailable","reason":"no-agent-connected"}`))
		return
	}
	defer unreg()
	for {
		_, data, err := c.Read(ctx)
		if err != nil {
			return
		}
		if a, ok := s.reg.AgentFor(user); ok {
			_ = a.Send(data)
		}
	}
}

package relaycmd

import (
	"fmt"
	"net/http"

	"github.com/spf13/cobra"

	"github.com/kslaboratory/kroot-adk/internal/relay"
)

func newServeCmd() *cobra.Command {
	var addr string
	cmd := &cobra.Command{
		Use:   "serve",
		Short: "Serve the ws relay (/ws/agent, /ws/browser)",
		RunE: func(cmd *cobra.Command, args []string) error {
			fmt.Fprintf(cmd.OutOrStderr(), "relay listening on %s\n", addr)
			return http.ListenAndServe(addr, relay.NewServer())
		},
	}
	cmd.Flags().StringVar(&addr, "addr", ":17070", "listen address")
	return cmd
}
