kroot-adk Chat — 데몬 agent(릴레이↔executor 브리지) (Option D2, Step 4) Implementation Plan ​

For agentic workers: REQUIRED SUB-SKILL: superpowers:subagent-driven-development. Steps use checkbox (- [ ]).

Goal: kroot-adk 데몬이 릴레이 /ws/agent 로 outbound ws 연결하고, 로컬 executor 와 무손실 바이트 브리지한다. 가짜 브라우저 → 릴레이 → 데몬(agent) → executor → claude → 역방향 end-to-end 를 (릴레이 in-process, PAT 없이) 실증한다. kroot chat start 로 기동.

핵심: 브라우저와 executor 는 같은 JSON(chat/permission_response/abort ↔ text/permission_request/done…)을 쓰므로, agent 는 파싱 없이 원본 라인을 그대로 중계한다:

  • 릴레이 ws → executor stdin (WriteRaw)
  • executor stdout(Event.Raw) → 릴레이 ws

Tech: Go 1.25 + github.com/coder/websocket(이미 추가됨). Node+claude(통합 테스트). 레포 github.com/kslaboratory/kroot-adk, 브랜치 feat/chat-connect-claude-driver.

// WriteRaw writes a raw NDJSON line to the executor's stdin (used by the relay
// agent to forward browser messages verbatim). A trailing newline is added.
func (e *Executor) WriteRaw(line []byte) error {
	e.writeMu.Lock()
	defer e.writeMu.Unlock()
	_, err := e.stdin.Write(append(append([]byte(nil), line...), '\n'))
	return err
}

package chatagent

import (
	"context"

	"os"
	"os/exec"

	"testing"
	"time"

// 가짜 브라우저 → 릴레이 → agent → executor → claude → 역방향. node/claude/deps 없으면 skip.
func TestAgent_EndToEnd(t *testing.T) {
	if _, err := exec.LookPath("node"); err != nil {
		t.Skip("node not on PATH")
	}
	if _, err := exec.LookPath("claude"); err != nil {
		t.Skip("claude not on PATH")
	}
	wd, _ := os.Getwd()
	repo := filepath.Clean(filepath.Join(wd, "..", ".."))
	exPath := filepath.Join(repo, "node-executor", "executor.mjs")
	if _, err := os.Stat(filepath.Join(repo, "node-executor", "node_modules")); err != nil {
		t.Skip("node-executor deps not installed")
	}

	srv := httptest.NewServer(relay.NewServer())
	defer srv.Close()
	base := "ws" + strings.TrimPrefix(srv.URL, "http")

	ctx, cancel := context.WithTimeout(context.Background(), 120*time.Second)
	defer cancel()

	// agent (daemon) connects to relay /ws/agent
	agentDone := make(chan error, 1)
	go func() { agentDone <- Run(ctx, base+"/ws/agent?user=u", exPath) }()
	time.Sleep(300 * time.Millisecond) // let agent register

	// fake browser connects
	browser, _, err := websocket.Dial(ctx, base+"/ws/browser?user=u", nil)
	if err != nil {
		t.Fatalf("browser dial: %v", err)
	}
	defer browser.Close(websocket.StatusNormalClosure, "")

	if err := browser.Write(ctx, websocket.MessageText,
		[]byte(`{"type":"chat","prompt":"Reply with exactly the word pong","cwd":"/tmp"}`)); err != nil {
		t.Fatalf("browser write: %v", err)
	}

	var sawText, sawDone bool
	for !sawDone {
		_, data, err := browser.Read(ctx)
		if err != nil {
			t.Fatalf("browser read: %v", err)
		}
		s := string(data)
		if strings.Contains(s, `"type":"error"`) {
			t.Fatalf("error from chain: %s", s)
		}
		if strings.Contains(s, `"type":"text"`) {
			sawText = true
		}
		if strings.Contains(s, `"type":"done"`) {
			sawDone = true
		}
	}
	if !sawText {
		t.Fatal("no text event reached the browser")
	}
	cancel()
	select {
	case <-agentDone:
	case <-time.After(3 * time.Second):
	}
}

// Package chatagent bridges the local executor to the relay's /ws/agent leg.
package chatagent

import (
	"context"

	"github.com/coder/websocket"

	"github.com/kslaboratory/kroot-adk/internal/executor"
)

// Run dials the relay agent endpoint, starts the executor, and bridges them
// (relay ws ⇄ executor stdio, verbatim) until ctx is done or either side
// closes. relayURL e.g. ws://host:17070/ws/agent?user=U.
func Run(ctx context.Context, relayURL, executorPath string) error {
	c, _, err := websocket.Dial(ctx, relayURL, nil)
	if err != nil {
		return err
	}
	defer c.Close(websocket.StatusNormalClosure, "")

	ex, err := executor.Start(ctx, "", executorPath)
	if err != nil {
		return err
	}
	defer ex.Close()

	errc := make(chan error, 2)

	// executor stdout → relay ws (raw)
	go func() {
		for ev := range ex.Events() {
			if len(ev.Raw) == 0 {
				continue
			}
			if err := c.Write(ctx, websocket.MessageText, ev.Raw); err != nil {
				errc <- err
				return
			}
		}
		errc <- nil
	}()

	// relay ws → executor stdin (raw)
	go func() {
		for {
			_, data, err := c.Read(ctx)
			if err != nil {
				errc <- err
				return
			}
			if err := ex.WriteRaw(data); err != nil {
				errc <- err
				return
			}
		}
	}()

	return <-errc
}

package chatcmd

import (
	"fmt"
	"net/url"
	"os/exec"

	"github.com/spf13/cobra"

	"github.com/kslaboratory/kroot-adk/internal/chatagent"
)

func newStartCmd() *cobra.Command {
	var (
		relayURL    string
		user        string
		executorDir string
	)
	cmd := &cobra.Command{
		Use:   "start",
		Short: "Start the chat daemon: connect to the relay and bridge the local executor",
		RunE: func(cmd *cobra.Command, args []string) error {
			if _, err := exec.LookPath("node"); err != nil {
				return fmt.Errorf("node를 찾을 수 없습니다. Node ≥ 20 을 설치하세요")
			}
			if relayURL == "" || user == "" {
				return fmt.Errorf("--relay 와 --user 가 필요합니다 (인증은 후속 PAT 단계)")
			}
			exPath, err := findExecutor(executorDir)
			if err != nil {
				return err
			}
			full := relayURL + "?user=" + url.QueryEscape(user)
			fmt.Fprintf(cmd.OutOrStderr(), "relay 연결: %s\n", full)
			return chatagent.Run(cmd.Context(), full, exPath)
		},
	}
	cmd.Flags().StringVar(&relayURL, "relay", "", "Relay agent ws URL (e.g. ws://host:17070/ws/agent)")
	cmd.Flags().StringVar(&user, "user", "", "User id (dev; replaced by PAT identity later)")
	cmd.Flags().StringVar(&executorDir, "executor-dir", "", "Directory containing executor.mjs (default: auto-detect)")
	return cmd
}
