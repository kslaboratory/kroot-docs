func TestPidfileRoundTrip(t *testing.T) {
	path := filepath.Join(t.TempDir(), "chat.pid")
	if err := WritePid(path, 4242); err != nil {
		t.Fatalf("WritePid: %v", err)
	}
	pid, err := ReadPid(path)
	if err != nil || pid != 4242 {
		t.Fatalf("ReadPid = %d, %v", pid, err)
	}
	if err := Remove(path); err != nil {
		t.Fatalf("Remove: %v", err)
	}
	if _, err := ReadPid(path); !os.IsNotExist(err) {
		t.Errorf("ReadPid after remove err = %v, want not-exist", err)
	}
}

func TestRunning(t *testing.T) {
	if !Running(os.Getpid()) {
		t.Error("Running(self) should be true")
	}
	if Running(0) || Running(-1) {
		t.Error("Running(0/-1) should be false")
	}
	// a very unlikely pid
	if Running(2147480000) {
		t.Error("Running(huge pid) should be false")
	}
}

func TestDefaultPidPath(t *testing.T) {
	t.Setenv("KROOT_CHAT_PIDFILE", "/custom/chat.pid")
	if p, _ := DefaultPidPath(); p != "/custom/chat.pid" {
		t.Errorf("DefaultPidPath env = %q", p)
	}
	t.Setenv("KROOT_CHAT_PIDFILE", "")
	p, err := DefaultPidPath()
	if err != nil {
		t.Fatal(err)
	}
	if filepath.Base(p) != "chat.pid" {
		t.Errorf("DefaultPidPath = %q", p)
	}
}

// Package chatdaemon manages the chat daemon's pidfile and liveness.
package chatdaemon

import (
	"os"
	"path/filepath"
	"strconv"
	"strings"
	"syscall"
)

// DefaultPidPath returns $KROOT_CHAT_PIDFILE or ~/.kroot/chat.pid.
func DefaultPidPath() (string, error) {
	if p := os.Getenv("KROOT_CHAT_PIDFILE"); p != "" {
		return p, nil
	}
	home, err := os.UserHomeDir()
	if err != nil {
		return "", err
	}
	return filepath.Join(home, ".kroot", "chat.pid"), nil
}

// WritePid writes pid to path (creating parent dirs).
func WritePid(path string, pid int) error {
	if err := os.MkdirAll(filepath.Dir(path), 0o755); err != nil {
		return err
	}
	return os.WriteFile(path, []byte(strconv.Itoa(pid)), 0o644)
}

// ReadPid reads the pid from path.
func ReadPid(path string) (int, error) {
	b, err := os.ReadFile(path)
	if err != nil {
		return 0, err
	}
	return strconv.Atoi(strings.TrimSpace(string(b)))
}

// Remove deletes the pidfile.
func Remove(path string) error { return os.Remove(path) }

// Running reports whether a process with pid is alive (signal 0 probe).
func Running(pid int) bool {
	if pid <= 0 {
		return false
	}
	p, err := os.FindProcess(pid)
	if err != nil {
		return false
	}
	return p.Signal(syscall.Signal(0)) == nil
}

//go:build !windows

package chatcmd

import (
	"os/exec"
	"syscall"
)

// detach configures cmd to start in its own session (background daemon).
func detach(cmd *exec.Cmd) {
	if cmd.SysProcAttr == nil {
		cmd.SysProcAttr = &syscall.SysProcAttr{}
	}
	cmd.SysProcAttr.Setsid = true
}

const daemonSupported = true

  • [ ] Step 3: stop — cmd/chatcmd/stop.go

go
package chatcmd

import (
	"fmt"
	"os"
	"syscall"

	"github.com/spf13/cobra"

	"github.com/kslaboratory/kroot-adk/internal/chatdaemon"
)

func newStopCmd() *cobra.Command {
	return &cobra.Command{
		Use:   "stop",
		Short: "Stop the chat daemon (SIGTERM to the pidfile process)",
		RunE: func(cmd *cobra.Command, args []string) error {
			path, err := chatdaemon.DefaultPidPath()
			if err != nil {
				return err
			}
			pid, err := chatdaemon.ReadPid(path)
			if err != nil {
				if os.IsNotExist(err) {
					fmt.Fprintln(cmd.OutOrStderr(), "실행 중인 데몬이 없습니다 (pidfile 없음)")
					return nil
				}
				return err
			}
			if !chatdaemon.Running(pid) {
				_ = chatdaemon.Remove(path)
				fmt.Fprintf(cmd.OutOrStderr(), "데몬(pid %d)이 없어 pidfile 정리\n", pid)
				return nil
			}
			p, err := os.FindProcess(pid)
			if err != nil {
				return err
			}
			if err := p.Signal(syscall.SIGTERM); err != nil {
				return fmt.Errorf("SIGTERM 실패: %w", err)
			}
			_ = chatdaemon.Remove(path)
			fmt.Fprintf(cmd.OutOrStderr(), "중지 요청(pid %d)\n", pid)
			return nil
		},
	}
}

func newStatusCmd() *cobra.Command {
	return &cobra.Command{
		Use:   "status",
		Short: "Show chat daemon status",
		RunE: func(cmd *cobra.Command, args []string) error {
			path, err := chatdaemon.DefaultPidPath()
			if err != nil {
				return err
			}
			pid, err := chatdaemon.ReadPid(path)
			if err != nil || !chatdaemon.Running(pid) {
				fmt.Fprintln(cmd.OutOrStderr(), "stopped")
				return nil
			}
			fmt.Fprintf(cmd.OutOrStderr(), "running (pid %d)\n", pid)
			return nil
		},
	}
}
