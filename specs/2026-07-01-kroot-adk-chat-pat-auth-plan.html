kroot-adk Chat — 릴레이 PAT 인증 (Option D2, Step 5) Implementation Plan ​

For agentic workers: REQUIRED SUB-SKILL: superpowers:subagent-driven-development. Steps use checkbox (- [ ]).

Goal: 릴레이의 agent leg(/ws/agent)를 PAT 로 인증한다. 데몬(kroot chat start)이 저장된 PAT 를 ws 핸드셰이크 Authorization: Bearer kpat_… 로 제시하면, 릴레이가 kroot 서버 gRPC WhoAmI 로 검증해 user id 를 얻어 그 user 로 agent 를 등록한다. dev ?user= 를 대체(브라우저 leg 인증은 Step 7).

설계: PAT 검증을 AgentAuthenticator 인터페이스 뒤로 추상화 → 릴레이는 백엔드 없이 단위 테스트(fake). 실제 검증은 krootapi.WhoAmIWithToken(c.cred 미접촉·per-call 토큰) 기반 impl. 인증은 opt-in — 미설정 시 기존 ?user= 유지(기존 e2e 불변).

Tech: Go 1.25 + 기존 coder/websocket·krootapi(gRPC). 레포 github.com/kslaboratory/kroot-adk, 브랜치 feat/chat-connect-claude-driver.

Backend(별도): server::auth PatService.ALLOWED_SCOPES 에 aicoding.connect 추가(사용자가 이 PAT 발급 가능하도록). — backend 서브모듈. 스코프 강제는 후속(WhoAmI 가 PAT 스코프를 반환하지 않으면 valid-PAT 식별로 충분).

go
// WhoAmIWithToken validates an arbitrary bearer token (e.g. an incoming agent's
// PAT) and returns its User identity — WITHOUT touching the client's stored
// credential. Server-side use (relay agent auth). Concurrency-safe (per-call ctx).
func (c *Client) WhoAmIWithToken(ctx context.Context, token string) (*krootv1.User, error) {
	md := metadata.Pairs("authorization", "Bearer "+token)
	ctx = metadata.NewOutgoingContext(ctx, md)
	resp, err := c.authRaw.WhoAmI(ctx, &krootv1.WhoAmIRequest{})
	if err != nil {
		if status.Code(err) == codes.Unauthenticated {
			return nil, ErrUnauthenticated
		}
		return nil, err
	}
	return resp.GetUser(), nil
}

필요 import: google.golang.org/grpc/metadata, google.golang.org/grpc/status, google.golang.org/grpc/codes (일부 이미 있음 — 중복 추가 금지).

package relay

import (
	"context"
	"errors"
	"net/http/httptest"
	"strings"
	"testing"
	"time"

	"github.com/coder/websocket"
)

type fakeAuth struct{ user string }

func (f fakeAuth) AgentUser(_ context.Context, token string) (string, error) {
	if token == "good" {
		return f.user, nil
	}
	return "", errors.New("bad token")
}

func TestServer_AgentPATAuth(t *testing.T) {
	srv := httptest.NewServer(NewServerWithAuth(fakeAuth{user: "alice"}))
	defer srv.Close()
	base := "ws" + strings.TrimPrefix(srv.URL, "http")
	ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
	defer cancel()

	// agent presents a good token via Authorization header → registered as "alice"
	agent, _, err := websocket.Dial(ctx, base+"/ws/agent", &websocket.DialOptions{
		HTTPHeader: map[string][]string{"Authorization": {"Bearer good"}},
	})
	if err != nil {
		t.Fatalf("agent dial: %v", err)
	}
	defer agent.Close(websocket.StatusNormalClosure, "")
	time.Sleep(100 * time.Millisecond)

	// browser binds as "alice" (dev ?user= on browser leg) → bridged
	browser, _, err := websocket.Dial(ctx, base+"/ws/browser?user=alice", nil)
	if err != nil {
		t.Fatalf("browser dial: %v", err)
	}
	defer browser.Close(websocket.StatusNormalClosure, "")
	if err := browser.Write(ctx, websocket.MessageText, []byte(`{"type":"chat","prompt":"x"}`)); err != nil {
		t.Fatalf("browser write: %v", err)
	}
	_, got, err := agent.Read(ctx)
	if err != nil {
		t.Fatalf("agent read: %v", err)
	}
	if !strings.Contains(string(got), `"prompt":"x"`) {
		t.Errorf("agent got %q", got)
	}
}

func TestServer_AgentBadToken(t *testing.T) {
	srv := httptest.NewServer(NewServerWithAuth(fakeAuth{user: "alice"}))
	defer srv.Close()
	base := "ws" + strings.TrimPrefix(srv.URL, "http")
	ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
	defer cancel()
	// no/invalid token → handshake rejected (Dial returns error)
	_, _, err := websocket.Dial(ctx, base+"/ws/agent", &websocket.DialOptions{
		HTTPHeader: map[string][]string{"Authorization": {"Bearer bad"}},
	})
	if err == nil {
		t.Fatal("expected agent dial to fail with bad token")
	}
}

package relay

import (
	"context"

	"github.com/kslaboratory/kroot-adk/internal/krootapi"
)

// PATAuthenticator validates agent PATs via the kroot server (WhoAmIWithToken).
type PATAuthenticator struct{ client *krootapi.Client }

// NewPATAuthenticator wraps a krootapi client.
func NewPATAuthenticator(c *krootapi.Client) *PATAuthenticator { return &PATAuthenticator{client: c} }

// AgentUser validates token and returns the user id.
func (p *PATAuthenticator) AgentUser(ctx context.Context, token string) (string, error) {
	u, err := p.client.WhoAmIWithToken(ctx, token)
	if err != nil {
		return "", err
	}
	return u.GetId(), nil
}
