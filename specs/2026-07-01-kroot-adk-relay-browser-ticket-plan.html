kroot-adk Relay — 브라우저 leg 티켓 인증 (Option D2, Step 7-T3) Implementation Plan ​

For agentic workers: REQUIRED SUB-SKILL: superpowers:subagent-driven-development. Steps use checkbox (- [ ]).

Goal: 릴레이 /ws/browser 를 HMAC 서명 티켓으로 인증한다. 프론트 BFF(세션 사용자 앎)가 공유 시크릿으로 <userId>.<exp>.<hmac> 티켓을 발급 → 브라우저가 ?ticket=… 로 접속 → 릴레이가 같은 시크릿으로 로컬 검증(백엔드 호출 없음) → userId. agent(PAT)·browser(티켓) 양쪽 user 가 일치할 때만 브리지.

설계(오류·의존 최소화): 티켓은 stateless HMAC(백엔드 신규 엔드포인트 불필요). 릴레이는 시크릿만 있으면 검증. 브라우저 인증은 opt-in(미설정 시 dev ?user= 유지 → 기존 테스트 불변).

Tech: Go 1.25 stdlib(crypto/hmac,crypto/sha256). 레포 github.com/kslaboratory/kroot-adk, 브랜치 feat/chat-connect-claude-driver.

티켓 포맷(TS/Go 상호운용): <userId>.<expUnix>.<hexHMACSHA256(secret, "<userId>.<expUnix>")>. userId=UUID(점 없음). 프론트 BFF 는 동일 포맷을 TS 로 발급.

import (
	"errors"

	"time"

func TestTicketRoundTrip(t *testing.T) {
	secret := []byte("shhh")
	tkt := MintTicket(secret, "user-abc", time.Minute)
	uid, err := VerifyTicket(secret, tkt)
	if err != nil || uid != "user-abc" {
		t.Fatalf("verify = %q, %v", uid, err)
	}
}

func TestTicketRejects(t *testing.T) {
	secret := []byte("shhh")
	good := MintTicket(secret, "user-abc", time.Minute)

	// wrong secret
	if _, err := VerifyTicket([]byte("other"), good); !errors.Is(err, ErrTicketInvalid) {
		t.Error("wrong secret should be invalid")
	}
	// tampered
	if _, err := VerifyTicket(secret, good+"x"); !errors.Is(err, ErrTicketInvalid) {
		t.Error("tampered should be invalid")
	}
	// malformed
	if _, err := VerifyTicket(secret, "nope"); !errors.Is(err, ErrTicketInvalid) {
		t.Error("malformed should be invalid")
	}
	// expired
	expired := MintTicket(secret, "user-abc", -time.Minute)
	if _, err := VerifyTicket(secret, expired); !errors.Is(err, ErrTicketInvalid) {
		t.Error("expired should be invalid")
	}
}

go
package relay

import (
	"crypto/hmac"
	"crypto/sha256"
	"encoding/hex"
	"errors"
	"strconv"
	"strings"
	"time"
)

// ErrTicketInvalid is returned for a malformed, tampered, or expired ticket.
var ErrTicketInvalid = errors.New("invalid ticket")

// MintTicket makes a short-lived browser ticket "<userID>.<expUnix>.<hexHMAC>".
// The frontend BFF mints this (in TS) with the shared secret for the session
// user; the same format/secret is verified here.
func MintTicket(secret []byte, userID string, ttl time.Duration) string {
	exp := time.Now().Add(ttl).Unix()
	msg := userID + "." + strconv.FormatInt(exp, 10)
	return msg + "." + sign(secret, msg)
}

func sign(secret []byte, msg string) string {
	m := hmac.New(sha256.New, secret)
	m.Write([]byte(msg))
	return hex.EncodeToString(m.Sum(nil))
}

// VerifyTicket checks the HMAC and expiry, returning the userID. The signature
// is the last dot-segment; the expiry the second-to-last; userID is everything
// before (so a userID may itself contain dots).
func VerifyTicket(secret []byte, ticket string) (string, error) {
	i2 := strings.LastIndex(ticket, ".")
	if i2 < 0 {
		return "", ErrTicketInvalid
	}
	sig := ticket[i2+1:]
	rest := ticket[:i2]
	i1 := strings.LastIndex(rest, ".")
	if i1 < 0 {
		return "", ErrTicketInvalid
	}
	expStr := rest[i1+1:]
	userID := rest[:i1]
	if userID == "" {
		return "", ErrTicketInvalid
	}
	if !hmac.Equal([]byte(sig), []byte(sign(secret, rest))) {
		return "", ErrTicketInvalid
	}
	exp, err := strconv.ParseInt(expStr, 10, 64)
	if err != nil {
		return "", ErrTicketInvalid
	}
	if time.Now().Unix() > exp {
		return "", ErrTicketInvalid
	}
	return userID, nil
}

package relay

import (
	"context"
	"net/http/httptest"
	"strings"
	"testing"
	"time"

	"github.com/coder/websocket"
)

func TestServer_BrowserTicketAuth(t *testing.T) {
	secret := []byte("shhh")
	srv := httptest.NewServer(NewServerOpts(ServerOptions{BrowserAuth: HMACTicketAuth{Secret: secret}}))
	defer srv.Close()
	base := "ws" + strings.TrimPrefix(srv.URL, "http")
	ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
	defer cancel()

	// agent registers as user "u" (dev ?user=, since AgentAuth not set)
	agent, _, err := websocket.Dial(ctx, base+"/ws/agent?user=u", nil)
	if err != nil {
		t.Fatalf("agent dial: %v", err)
	}
	defer agent.Close(websocket.StatusNormalClosure, "")
	time.Sleep(100 * time.Millisecond)

	// browser presents a valid ticket for user "u" → bridged
	tkt := MintTicket(secret, "u", time.Minute)
	browser, _, err := websocket.Dial(ctx, base+"/ws/browser?ticket="+tkt, nil)
	if err != nil {
		t.Fatalf("browser dial: %v", err)
	}
	defer browser.Close(websocket.StatusNormalClosure, "")
	if err := browser.Write(ctx, websocket.MessageText, []byte(`{"type":"chat","prompt":"y"}`)); err != nil {
		t.Fatalf("browser write: %v", err)
	}
	_, got, err := agent.Read(ctx)
	if err != nil {
		t.Fatalf("agent read: %v", err)
	}
	if !strings.Contains(string(got), `"prompt":"y"`) {
		t.Errorf("agent got %q", got)
	}
}

func TestServer_BrowserBadTicket(t *testing.T) {
	srv := httptest.NewServer(NewServerOpts(ServerOptions{BrowserAuth: HMACTicketAuth{Secret: []byte("shhh")}}))
	defer srv.Close()
	base := "ws" + strings.TrimPrefix(srv.URL, "http")
	ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
	defer cancel()
	_, _, err := websocket.Dial(ctx, base+"/ws/browser?ticket=bogus", nil)
	if err == nil {
		t.Fatal("expected browser dial to fail with bad ticket")
	}
}
