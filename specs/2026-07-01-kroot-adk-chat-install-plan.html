Goal: kroot chat install — Node 실행기 소스(executor.mjs, package.json)를 Go 바이너리에 go:embed로 번들하고, 실행 시 ~/.kroot/chat-executor/ 로 풀어 npm install 한다. 이후 kroot chat run(및 향후 start/stop)이 레포 경로가 아니라 설치된 실행기로 동작한다.

Why first: 이후 모든 단계(start/stop·릴레이 데몬)가 "설치된 실행기"에 의존한다. 네트워크·인증 없이 kroot-adk 안에서만 끝나는 저위험 단계.

Tech: Go 1.25(embed), Node ≥ 20 + npm(사용자 설치). 레포 github.com/kslaboratory/kroot-adk, 브랜치 feat/chat-connect-claude-driver.

package chatinstall

import (
	"os"

func TestWriteSources(t *testing.T) {
	dir := t.TempDir()
	if err := WriteSources(dir); err != nil {
		t.Fatalf("WriteSources: %v", err)
	}
	for _, name := range []string{"executor.mjs", "package.json"} {
		b, err := os.ReadFile(filepath.Join(dir, name))
		if err != nil {
			t.Fatalf("missing %s: %v", name, err)
		}
		if len(b) == 0 {
			t.Errorf("%s is empty", name)
		}
	}
	// executor.mjs should look like our executor (sanity)
	b, _ := os.ReadFile(filepath.Join(dir, "executor.mjs"))
	if !containsAll(string(b), "claude-agent-sdk", "permission_request") {
		t.Errorf("executor.mjs content unexpected")
	}
}

func TestDefaultDir(t *testing.T) {
	t.Setenv("KROOT_CHAT_EXECUTOR_DIR", "/custom/dir")
	if d, _ := DefaultDir(); d != "/custom/dir" {
		t.Errorf("DefaultDir with env = %q", d)
	}
	t.Setenv("KROOT_CHAT_EXECUTOR_DIR", "")
	d, err := DefaultDir()
	if err != nil {
		t.Fatal(err)
	}
	if filepath.Base(d) != "chat-executor" {
		t.Errorf("DefaultDir = %q, want .../chat-executor", d)
	}
}

func containsAll(s string, subs ...string) bool {
	for _, sub := range subs {
		found := false
		for i := 0; i+len(sub) <= len(s); i++ {
			if s[i:i+len(sub)] == sub {
				found = true
				break
			}
		}
		if !found {
			return false
		}
	}
	return true
}

// Package chatinstall installs the embedded Node chat executor to the user's
// machine (~/.kroot/chat-executor) and runs npm install.
package chatinstall

import (
	"fmt"
	"io"
	"io/fs"
	"os"
	"os/exec"
	"path/filepath"

	nodeexecutor "github.com/kslaboratory/kroot-adk/node-executor"
)

// DefaultDir returns $KROOT_CHAT_EXECUTOR_DIR or ~/.kroot/chat-executor.
func DefaultDir() (string, error) {
	if env := os.Getenv("KROOT_CHAT_EXECUTOR_DIR"); env != "" {
		return env, nil
	}
	home, err := os.UserHomeDir()
	if err != nil {
		return "", err
	}
	return filepath.Join(home, ".kroot", "chat-executor"), nil
}

// WriteSources writes the embedded executor sources into dir (creating it).
func WriteSources(dir string) error {
	if err := os.MkdirAll(dir, 0o755); err != nil {
		return err
	}
	return fs.WalkDir(nodeexecutor.Files, ".", func(p string, d fs.DirEntry, err error) error {
		if err != nil || d.IsDir() {
			return err
		}
		b, err := nodeexecutor.Files.ReadFile(p)
		if err != nil {
			return err
		}
		return os.WriteFile(filepath.Join(dir, p), b, 0o644)
	})
}

// Install writes sources into dir and runs `npm install` there. Requires node+npm.
// Progress is written to out/errOut.
func Install(dir string, out, errOut io.Writer) error {
	if _, err := exec.LookPath("node"); err != nil {
		return fmt.Errorf("node를 찾을 수 없습니다. Node ≥ 20 을 설치하세요")
	}
	npm, err := exec.LookPath("npm")
	if err != nil {
		return fmt.Errorf("npm을 찾을 수 없습니다. Node.js(npm 포함)를 설치하세요")
	}
	if err := WriteSources(dir); err != nil {
		return fmt.Errorf("실행기 소스 기록 실패: %w", err)
	}
	cmd := exec.Command(npm, "install", "--no-audit", "--no-fund")
	cmd.Dir = dir
	cmd.Stdout = out
	cmd.Stderr = errOut
	if err := cmd.Run(); err != nil {
		return fmt.Errorf("npm install 실패: %w", err)
	}
	return nil
}

package chatcmd

import (
	"fmt"
	"os"

	"github.com/spf13/cobra"

	"github.com/kslaboratory/kroot-adk/internal/chatinstall"
)

func newInstallCmd() *cobra.Command {
	var dir string
	cmd := &cobra.Command{
		Use:   "install",
		Short: "Install the node chat executor to ~/.kroot/chat-executor (npm install)",
		RunE: func(cmd *cobra.Command, args []string) error {
			target := dir
			if target == "" {
				d, err := chatinstall.DefaultDir()
				if err != nil {
					return err
				}
				target = d
			}
			fmt.Fprintf(os.Stderr, "chat executor 설치 중: %s\n", target)
			if err := chatinstall.Install(target, os.Stdout, os.Stderr); err != nil {
				return err
			}
			fmt.Fprintf(os.Stderr, "완료: %s\n", target)
			return nil
		},
	}
	cmd.Flags().StringVar(&dir, "dir", "", "Install directory (default: ~/.kroot/chat-executor)")
	return cmd
}
