// Package chat drives the local `claude` Claude Code CLI in headless
// (stream-json) mode and exposes its output as typed events.
package chat

import (
	"bytes"
	"encoding/json"
)

// EventKind is the category of a parsed stream-json event.
type EventKind string

const (
	EventSystemInit EventKind = "system_init" // carries SessionID
	EventText       EventKind = "text"
	EventThinking   EventKind = "thinking"
	EventResult     EventKind = "result" // terminal; carries final text + SessionID
	EventError      EventKind = "error"  // transport/parse/process error
)

// Event is one parsed unit of claude CLI stream-json output.
type Event struct {
	Kind      EventKind
	Text      string // text/thinking body, or result text
	SessionID string // system_init / result
	Err       string // EventError
}

// ParseLine parses a single stream-json line into zero or more events.
// An assistant line may carry multiple content blocks → multiple events.
// Unrecognized or non-JSON lines yield nil (ignored).
func ParseLine(line []byte) []Event {
	line = bytes.TrimSpace(line)
	if len(line) == 0 {
		return nil
	}
	var head struct {
		Type      string `json:"type"`
		SessionID string `json:"session_id"`
		Result    string `json:"result"`
	}
	if err := json.Unmarshal(line, &head); err != nil {
		return nil
	}
	switch head.Type {
	case "system":
		return []Event{{Kind: EventSystemInit, SessionID: head.SessionID}}
	case "result":
		return []Event{{Kind: EventResult, Text: head.Result, SessionID: head.SessionID}}
	case "assistant":
		var m struct {
			Message struct {
				Content []struct {
					Type     string `json:"type"`
					Text     string `json:"text"`
					Thinking string `json:"thinking"`
				} `json:"content"`
			} `json:"message"`
		}
		if err := json.Unmarshal(line, &m); err != nil {
			return nil
		}
		var out []Event
		for _, b := range m.Message.Content {
			switch b.Type {
			case "text":
				out = append(out, Event{Kind: EventText, Text: b.Text})
			case "thinking":
				out = append(out, Event{Kind: EventThinking, Text: b.Thinking})
			}
		}
		return out
	}
	return nil
}

func TestStreamEvents(t *testing.T) {
	in := strings.Join([]string{
		`{"type":"system","subtype":"init","session_id":"s1"}`,
		``,                       // blank line ignored
		`garbage`,                // invalid ignored
		`{"type":"assistant","message":{"content":[{"type":"text","text":"Hi"}]}}`,

	var got []Event
	for ev := range StreamEvents(strings.NewReader(in)) {
		got = append(got, ev)

		if got[i].Kind != want[i] {

		}
	}
	if got[0].SessionID != "s1" {
		t.Errorf("system_init SessionID = %q, want s1", got[0].SessionID)
	}
}

package chat

import (
	"bufio"
	"io"
)

// maxLine bounds a single stream-json line (large tool outputs).
const maxLine = 8 * 1024 * 1024

// StreamEvents reads line-delimited stream-json from r and emits parsed
// events on the returned channel, which is closed at EOF.
func StreamEvents(r io.Reader) <-chan Event {
	ch := make(chan Event)
	go func() {
		defer close(ch)
		sc := bufio.NewScanner(r)
		sc.Buffer(make([]byte, 0, 64*1024), maxLine)
		for sc.Scan() {
			for _, ev := range ParseLine(sc.Bytes()) {
				ch <- ev
			}
		}
	}()
	return ch
}

package chat

// BuildArgs renders Options into claude CLI arguments for a headless
// stream-json run.
func BuildArgs(o Options) []string {
	args := []string{"-p", o.Prompt, "--output-format", "stream-json", "--verbose"}
	if o.Model != "" {
		args = append(args, "--model", o.Model)
	}
	if o.PermissionMode != "" {
		args = append(args, "--permission-mode", o.PermissionMode)
	}
	switch {
	case o.Resume != "":
		args = append(args, "--resume", o.Resume)
	case o.SessionID != "":
		args = append(args, "--session-id", o.SessionID)
	}
	return args
}

package chat

import (
	"context"
	"os/exec"
)

// Client runs a single claude headless invocation.
type Client struct {
	opts Options
}

// New returns a Client for the given options.
func New(o Options) *Client { return &Client{opts: o} }

// Run spawns claude with the configured options and returns a channel of
// parsed events, closed when the process exits. A non-zero exit or stdout
// wiring failure is surfaced as a terminal EventError on the channel.
func (c *Client) Run(ctx context.Context) (<-chan Event, error) {
	bin := c.opts.ClaudePath
	if bin == "" {
		bin = "claude"
	}
	cmd := exec.CommandContext(ctx, bin, BuildArgs(c.opts)...)
	if c.opts.Cwd != "" {
		cmd.Dir = c.opts.Cwd
	}
	stdout, err := cmd.StdoutPipe()
	if err != nil {
		return nil, err
	}
	if err := cmd.Start(); err != nil {
		return nil, err
	}
	out := make(chan Event)
	go func() {
		defer close(out)
		for ev := range StreamEvents(stdout) {
			out <- ev
		}
		if err := cmd.Wait(); err != nil {
			out <- Event{Kind: EventError, Err: err.Error()}
		}
	}()
	return out, nil
}

package chatcmd

import (
	"context"
	"fmt"
	"os"

	"github.com/spf13/cobra"

	"github.com/kslaboratory/kroot-adk/internal/chat"
)

func newRunCmd() *cobra.Command {
	var (
		cwd            string
		model          string
		permissionMode string
	)
	cmd := &cobra.Command{
		Use:   "run <prompt>",
		Short: "Run a one-shot Claude Code prompt locally (driver smoke)",
		Args:  cobra.MinimumNArgs(1),
		RunE: func(cmd *cobra.Command, args []string) error {
			prompt := args[0]
			c := chat.New(chat.Options{
				Prompt:         prompt,
				Cwd:            cwd,
				Model:          model,
				PermissionMode: permissionMode,
			})
			ch, err := c.Run(context.Background())
			if err != nil {
				return fmt.Errorf("claude 실행 실패: %w", err)
			}
			for ev := range ch {
				switch ev.Kind {
				case chat.EventText, chat.EventResult:
					fmt.Print(ev.Text)
				case chat.EventError:
					fmt.Fprintf(os.Stderr, "\n[error] %s\n", ev.Err)
				}
			}
			fmt.Println()
			return nil
		},
	}
	cmd.Flags().StringVar(&cwd, "cwd", "", "Project working directory (loads its .claude/ + CLAUDE.md)")
	cmd.Flags().StringVar(&model, "model", "", "Model id (e.g. claude-sonnet-4-6)")
	cmd.Flags().StringVar(&permissionMode, "permission-mode", "", "acceptEdits|bypassPermissions|default")
	return cmd
}
