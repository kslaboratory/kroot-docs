package chat

import "encoding/json"

// ControlKind classifies a control_* message the CLI sends us on stdout.
type ControlKind string

const (
	ControlCanUseTool ControlKind = "can_use_tool"
	ControlCancel     ControlKind = "cancel" // control_cancel_request
)

// PermissionRequest is a can_use_tool control_request (CLI → host).
// Protocol: findings §2.
type PermissionRequest struct {
	RequestID string          // echo into control_response
	ToolName  string          // e.g. "Write", "Bash"
	ToolUseID string          // echo as toolUseID in response
	Input     json.RawMessage // tool arguments
}

// ControlMessage is a parsed control message from the CLI's stdout.
type ControlMessage struct {
	Kind       ControlKind
	Permission PermissionRequest // Kind == ControlCanUseTool
	CancelID   string            // Kind == ControlCancel
}

// ParseControl reports whether line is an inbound control message we handle
// (can_use_tool request, or cancel of a pending request). Data events and
// outbound-only kinds (e.g. interrupt) return ok=false.
func ParseControl(line []byte) (ControlMessage, bool) {
	var head struct {
		Type      string `json:"type"`
		RequestID string `json:"request_id"`
		Request   struct {
			Subtype   string          `json:"subtype"`
			ToolName  string          `json:"tool_name"`
			ToolUseID string          `json:"tool_use_id"`
			Input     json.RawMessage `json:"input"`
		} `json:"request"`
	}
	if err := json.Unmarshal(line, &head); err != nil {
		return ControlMessage{}, false
	}
	switch head.Type {
	case "control_request":
		if head.Request.Subtype == "can_use_tool" {
			return ControlMessage{Kind: ControlCanUseTool, Permission: PermissionRequest{
				RequestID: head.RequestID,
				ToolName:  head.Request.ToolName,
				ToolUseID: head.Request.ToolUseID,
				Input:     head.Request.Input,
			}}, true
		}
		return ControlMessage{}, false
	case "control_cancel_request":
		return ControlMessage{Kind: ControlCancel, CancelID: head.RequestID}, true
	}
	return ControlMessage{}, false
}

package chat

import "encoding/json"

// PermissionDecision is the host's answer to a can_use_tool request (findings §3).
type PermissionDecision struct {
	Allow        bool
	UpdatedInput json.RawMessage // allow only; optional modified tool args
	Message      string          // deny reason (tool_result error)
	Interrupt    bool            // deny: also abort the turn
}

type userMsg struct {
	Type            string      `json:"type"`
	Message         userPayload `json:"message"`
	ParentToolUseID *string     `json:"parent_tool_use_id"`
}
type userPayload struct {
	Role    string `json:"role"`
	Content string `json:"content"`
}

// EncodeUserMessage builds a stdin "user" message (findings §1).
func EncodeUserMessage(prompt string) ([]byte, error) {
	return json.Marshal(userMsg{
		Type:    "user",
		Message: userPayload{Role: "user", Content: prompt},
		// ParentToolUseID nil → encodes as JSON null (required field).
	})
}

type ctrlRespInner struct {
	Behavior     string          `json:"behavior"`
	ToolUseID    string          `json:"toolUseID"`
	UpdatedInput json.RawMessage `json:"updatedInput,omitempty"`
	Message      string          `json:"message,omitempty"`
	Interrupt    bool            `json:"interrupt,omitempty"`
}
type ctrlRespOuter struct {
	Type     string          `json:"type"`
	Response ctrlRespSuccess `json:"response"`
}
type ctrlRespSuccess struct {
	Subtype   string        `json:"subtype"`
	RequestID string        `json:"request_id"`
	Response  ctrlRespInner `json:"response"`
}

// EncodeControlResponse builds a control_response to a can_use_tool request
// (findings §3). reqID echoes the request_id; toolUseID echoes tool_use_id.
func EncodeControlResponse(reqID, toolUseID string, d PermissionDecision) ([]byte, error) {
	inner := ctrlRespInner{ToolUseID: toolUseID}
	if d.Allow {
		inner.Behavior = "allow"
		inner.UpdatedInput = d.UpdatedInput
	} else {
		inner.Behavior = "deny"
		inner.Message = d.Message
		inner.Interrupt = d.Interrupt
	}
	return json.Marshal(ctrlRespOuter{
		Type:     "control_response",
		Response: ctrlRespSuccess{Subtype: "success", RequestID: reqID, Response: inner},
	})
}

type interruptReq struct {
	Type      string             `json:"type"`
	RequestID string             `json:"request_id"`
	Request   interruptReqDetail `json:"request"`
}
type interruptReqDetail struct {
	Subtype string `json:"subtype"`
}

// EncodeInterrupt builds an interrupt control_request (findings §4).
func EncodeInterrupt(reqID string) ([]byte, error) {
	return json.Marshal(interruptReq{
		Type: "control_request", RequestID: reqID,
		Request: interruptReqDetail{Subtype: "interrupt"},
	})
}

// BuildInteractiveArgs renders Options for a BIDIRECTIONAL stream-json session
// (findings: --input-format stream-json, no -p; --verbose required). Prompt is
// sent over stdin, not as a flag.
func BuildInteractiveArgs(o Options) []string {
	args := []string{"--output-format", "stream-json", "--verbose", "--input-format", "stream-json"}
	if o.Model != "" {
		args = append(args, "--model", o.Model)
	}
	if o.PermissionMode != "" {
		args = append(args, "--permission-mode", o.PermissionMode)
	}
	switch {
	case o.Resume != "":
		args = append(args, "--resume", o.Resume)
	case o.SessionID != "":
		args = append(args, "--session-id", o.SessionID)
	}
	return args
}

func TestSessionPermissionFlow(t *testing.T) {
	ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
	defer cancel()

	// 가짜 CLI stdout 스크립트: init → assistant text → can_use_tool → (호스트 allow 대기) → result
	stdout := strings.Join([]string{
		`{"type":"system","subtype":"init","session_id":"sess-1"}`,
		`{"type":"assistant","message":{"content":[{"type":"text","text":"working"}]}}`,
		`{"type":"control_request","request_id":"req-1","request":{"subtype":"can_use_tool","tool_name":"Write","input":{"file_path":"/tmp/x"},"tool_use_id":"toolu_9"}}`,
		`{"type":"result","subtype":"success","session_id":"sess-1","result":"done"}`,
	}, "\n") + "\n"

	stdinBuf := &syncBuffer{}
	var gotReq PermissionRequest
	perm := func(req PermissionRequest) PermissionDecision {
		gotReq = req
		return PermissionDecision{Allow: true}
	}

	sess := newSession(stdinBuf, perm)
	events := sess.Events()
	go sess.runLoop(ctx, strings.NewReader(stdout))

	var kinds []EventKind
	for ev := range events {
		kinds = append(kinds, ev.Kind)
	}

	// 이벤트: system_init, text, result
	if len(kinds) != 3 || kinds[0] != EventSystemInit || kinds[1] != EventText || kinds[2] != EventResult {
		t.Fatalf("events = %v", kinds)
	}
	if sess.SessionID() != "sess-1" {
		t.Errorf("SessionID = %q", sess.SessionID())
	}
	// 권한 콜백이 올바른 요청을 받았는가
	if gotReq.RequestID != "req-1" || gotReq.ToolName != "Write" || gotReq.ToolUseID != "toolu_9" {
		t.Errorf("perm req = %+v", gotReq)
	}
	// stdin 에 allow control_response 가 기록됐는가
	wrote := stdinBuf.String()
	dec := json.NewDecoder(bufio.NewReader(strings.NewReader(wrote)))
	var found bool
	for {
		var m map[string]any
		if err := dec.Decode(&m); err == io.EOF {
			break
		} else if err != nil {
			break
		}
		if m["type"] == "control_response" {
			resp := m["response"].(map[string]any)
			if resp["request_id"] == "req-1" {
				inner := resp["response"].(map[string]any)
				if inner["behavior"] == "allow" && inner["toolUseID"] == "toolu_9" {
					found = true
				}
			}
		}
	}
	if !found {
		t.Errorf("no allow control_response written to stdin; got: %s", wrote)
	}
}

package chat

import (
	"bufio"
	"context"
	"crypto/rand"
	"encoding/hex"
	"io"
	"sync"
)

// PermissionFunc decides a can_use_tool request. Called from the session's
// read loop; it should not block indefinitely.
type PermissionFunc func(PermissionRequest) PermissionDecision

// Session drives one bidirectional claude stream-json conversation. Construct
// via StartSession (spawns claude) or newSession (for tests over pipes).
type Session struct {
	stdin io.Writer
	perm  PermissionFunc

	events chan Event

	writeMu sync.Mutex // serialize stdin writes (single-writer invariant)

	mu        sync.Mutex
	sessionID string
}

func newSession(stdin io.Writer, perm PermissionFunc) *Session {
	if perm == nil {
		perm = func(PermissionRequest) PermissionDecision {
			return PermissionDecision{Allow: false, Message: "no permission handler"}
		}
	}
	return &Session{stdin: stdin, perm: perm, events: make(chan Event)}
}

// Events returns the channel of data events; closed when the read loop ends.
func (s *Session) Events() <-chan Event { return s.events }

// SessionID returns the claude session id once the system/init message arrives.
func (s *Session) SessionID() string {
	s.mu.Lock()
	defer s.mu.Unlock()
	return s.sessionID
}

// writeLine writes one NDJSON line to stdin under the single-writer lock.
func (s *Session) writeLine(b []byte) error {
	s.writeMu.Lock()
	defer s.writeMu.Unlock()
	if _, err := s.stdin.Write(append(b, '\n')); err != nil {
		return err
	}
	return nil
}

// Send writes a user message (findings §1). Safe for multi-turn.
func (s *Session) Send(prompt string) error {
	b, err := EncodeUserMessage(prompt)
	if err != nil {
		return err
	}
	return s.writeLine(b)
}

// Interrupt aborts the in-progress turn (findings §4).
func (s *Session) Interrupt() error {
	b, err := EncodeInterrupt(randID("irq"))
	if err != nil {
		return err
	}
	return s.writeLine(b)
}

// runLoop reads stdout NDJSON: data events → Events channel; can_use_tool →
// PermissionFunc → control_response on stdin; control_cancel_request → ignored
// (handler is synchronous here, nothing pending to cancel). Closes Events at EOF.
// ctx cancellation aborts channel sends (M-2).
func (s *Session) runLoop(ctx context.Context, stdout io.Reader) {
	defer close(s.events)
	sc := bufio.NewScanner(stdout)
	sc.Buffer(make([]byte, 0, 64*1024), maxLine)
	for sc.Scan() {
		line := sc.Bytes()
		if cm, ok := ParseControl(line); ok {
			switch cm.Kind {
			case ControlCanUseTool:
				s.mu.Lock()
				// capture nothing; handle inline
				s.mu.Unlock()
				d := s.perm(cm.Permission)
				resp, err := EncodeControlResponse(cm.Permission.RequestID, cm.Permission.ToolUseID, d)
				if err == nil {
					_ = s.writeLine(resp)
				}
			case ControlCancel:
				// synchronous handler → nothing pending; ignore.
			}
			continue
		}
		for _, ev := range ParseLine(line) {
			if ev.Kind == EventSystemInit && ev.SessionID != "" {
				s.mu.Lock()
				s.sessionID = ev.SessionID
				s.mu.Unlock()
			}
			select {
			case s.events <- ev:
			case <-ctx.Done():
				return
			}
		}
	}
}

func randID(prefix string) string {
	var b [8]byte
	_, _ = rand.Read(b[:])
	return prefix + "-" + hex.EncodeToString(b[:])
}

go
package chat

import (
	"context"
	"os/exec"
)

// StartSession spawns claude in bidirectional stream-json mode, wires stdin/
// stdout, starts the read loop, and returns a ready Session. Caller drives it
// via Send/Interrupt and consumes Events(); the session ends when claude exits
// (Events closed). stdin is held open for the session lifetime (findings §6).
func StartSession(ctx context.Context, o Options, perm PermissionFunc) (*Session, error) {
	bin := o.ClaudePath
	if bin == "" {
		bin = "claude"
	}
	cmd := exec.CommandContext(ctx, bin, BuildInteractiveArgs(o)...)
	if o.Cwd != "" {
		cmd.Dir = o.Cwd
	}
	stdin, err := cmd.StdinPipe()
	if err != nil {
		return nil, err
	}
	stdout, err := cmd.StdoutPipe()
	if err != nil {
		return nil, err
	}
	if err := cmd.Start(); err != nil {
		return nil, err
	}
	sess := newSession(stdin, perm)
	go func() {
		sess.runLoop(ctx, stdout)
		// read loop ended (claude closed stdout). Close stdin to signal EOF and
		// reap the process.
		_ = stdin.Close()
		_ = cmd.Wait()
	}()
	return sess, nil
}

package chatcmd

import (
	"bufio"
	"context"
	"fmt"
	"os"
	"strings"

	"github.com/spf13/cobra"

	"github.com/kslaboratory/kroot-adk/internal/chat"
)

func newRunCmd() *cobra.Command {
	var (
		cwd            string
		model          string
		permissionMode string
		autoApprove    bool
	)
	cmd := &cobra.Command{
		Use:   "run <prompt>",
		Short: "Run a Claude Code prompt locally (bidirectional driver smoke)",
		Args:  cobra.MinimumNArgs(1),
		RunE: func(cmd *cobra.Command, args []string) error {
			prompt := strings.Join(args, " ")
			if permissionMode == "" {
				permissionMode = "default"
			}
			in := bufio.NewReader(os.Stdin)
			perm := func(req chat.PermissionRequest) chat.PermissionDecision {
				if autoApprove {
					return chat.PermissionDecision{Allow: true}
				}
				fmt.Fprintf(os.Stderr, "\n[권한] %s 실행을 허용할까요? (y/N) ", req.ToolName)
				line, _ := in.ReadString('\n')
				if strings.EqualFold(strings.TrimSpace(line), "y") {
					return chat.PermissionDecision{Allow: true}
				}
				return chat.PermissionDecision{Allow: false, Message: "사용자가 거부함"}
			}

			sess, err := chat.StartSession(cmd.Context(), chat.Options{
				Cwd: cwd, Model: model, PermissionMode: permissionMode,
			}, perm)
			if err != nil {
				return fmt.Errorf("claude 세션 시작 실패: %w", err)
			}
			if err := sess.Send(prompt); err != nil {
				return fmt.Errorf("프롬프트 전송 실패: %w", err)
			}
			for ev := range sess.Events() {
				switch ev.Kind {
				case chat.EventText:
					fmt.Print(ev.Text)
				case chat.EventError:
					fmt.Fprintf(os.Stderr, "\n[error] %s\n", ev.Err)
				}
			}
			fmt.Println()
			return nil
		},
	}
	cmd.Flags().StringVar(&cwd, "cwd", "", "Project working directory (loads its .claude/ + CLAUDE.md)")
	cmd.Flags().StringVar(&model, "model", "", "Model id (e.g. claude-sonnet-4-6)")
	cmd.Flags().StringVar(&permissionMode, "permission-mode", "", "default|acceptEdits|bypassPermissions (default: default)")
	cmd.Flags().BoolVar(&autoApprove, "yes", false, "Auto-approve all tool permission requests")
	return cmd
}
