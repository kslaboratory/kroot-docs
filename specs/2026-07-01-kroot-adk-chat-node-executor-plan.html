kroot-adk Chat — Node 실행기 (Option D2, Phase A) Implementation Plan ​

For agentic workers: REQUIRED SUB-SKILL: superpowers:subagent-driven-development. Steps use checkbox (- [ ]).

Goal: kroot-adk 레포 안에 stdio 기반 Node 실행기(node-executor/executor.mjs)를 만든다. stdin 으로 NDJSON chat 요청을 받아 @anthropic-ai/claude-agent-sdk query()(→ claude CLI 헤드리스 spawn)를 구동하고, 이벤트를 stdout NDJSON 으로 낸다. 네트워크 없음 — 이후 Go(kroot-adk)가 stdio ↔ 릴레이를 중계한다.

Architecture (Option D2):

[브라우저] ─wss─ [서버 릴레이] ─gRPC(PAT)─ [kroot-adk(Go)] ─stdio(NDJSON)─ [node executor.mjs] ─SDK─► claude
  • Go = 릴레이 연결(PAT)·감독·중계. Node executor = SDK 실행 어댑터(네트워크·인증 없음). SDK 가 claude CLI spawn + 양방향/권한 프로토콜 유지보수.
  • 본 Phase A = Node executor 만 (Go start/stop 은 Phase B, 릴레이는 Phase C).

출처: kroot-main-site/server.ts handleClaudeMessage(lines ~359–546)를 stdio 로 충실 이식. 결정 근거: 2026-06-30-kroot-adk-chat-connect-design.md, 2026-06-30-claude-control-protocol-findings.md, 그리고 Option D 전환(커밋 b46eb4c).

Tech: Node ≥ 20(ESM), @anthropic-ai/claude-agent-sdk@^0.3.150. Node 는 사용자 설치 전제. 레포 github.com/kslaboratory/kroot-adk, 브랜치 feat/chat-connect-claude-driver.

stdio 프로토콜 (Go ↔ Node, NDJSON 한 줄당 한 메시지 + \n) ​

요청 (Go → executor stdin):

jsonc
{"type":"chat","prompt":"...","cwd":"/abs/path","model":"claude-sonnet-4-6","permissionMode":"default","sessionId":null,"extendedThinking":false,"claudePath":null}
{"type":"permission_response","requestId":"<echo>","allow":true,"updatedInput":null}
{"type":"abort"}

이벤트 (executor stdout → Go):

jsonc
{"type":"ready"}                                  // 기동 완료
{"type":"session_id","sessionId":"<uuid>"}
{"type":"text","text":"..."}
{"type":"thinking","text":"..."}
{"type":"tool_call","name":"Write","input":{...}}
{"type":"tool_result","content":...}
{"type":"permission_request","requestId":"<uuid>","toolName":"Write","input":{...}}
{"type":"done","sessionId":"<uuid|null>"}
{"type":"error","message":"..."}
  • 권한: executor 가 permission_request 를 내보내고, 같은 requestId 의 permission_response 를 기다림(pending Map). chat 처리는 await 하지 않음 — stdin 이 계속 열려 permission_response 수신 가능해야 함.

#!/usr/bin/env node
// kroot-adk chat executor — stdio adapter around @anthropic-ai/claude-agent-sdk.
//
// Reads NDJSON requests on stdin, drives query() (which spawns the claude CLI
// headless), and writes NDJSON events on stdout. No network: the Go supervisor
// bridges this stdio to the relay. Ported from kroot-main-site server.ts
// handleClaudeMessage (session_id capture, stream_event deltas, assistant
// tool_use, user tool_result, result→done, canUseTool permission callback).

import { createInterface } from 'node:readline'
import { existsSync } from 'node:fs'
import os from 'node:os'
import { randomUUID } from 'node:crypto'

const DEFAULT_MODEL = 'claude-sonnet-4-6'

// out serializes one NDJSON event to stdout.
const out = (obj) => process.stdout.write(JSON.stringify(obj) + '\n')

// requestId -> resolver for pending permission decisions.
const pendingPermissions = new Map()
// the in-flight query (for abort/interrupt).
let currentQuery = null

function buildOptions(req) {
  let cwd = req.cwd
  if (!cwd || !existsSync(cwd)) cwd = os.homedir()

  const options = {
    cwd,
    permissionMode: req.permissionMode || 'default',
    model: req.model || DEFAULT_MODEL,
    // parity: load user+project settings so skills/subagents/hooks/commands work.
    settingSources: ['user', 'project'],
    // real-time token streaming (stream_event deltas).
    includePartialMessages: true,
  }
  if (req.claudePath) options.pathToClaudeCodeExecutable = req.claudePath
  if (req.extendedThinking) options.thinking = { type: 'enabled', budget_tokens: 10000 }
  if (req.sessionId) options.resume = req.sessionId

  if (options.permissionMode !== 'bypassPermissions') {
    options.canUseTool = (toolName, input) =>
      new Promise((resolve) => {
        const requestId = randomUUID()
        pendingPermissions.set(requestId, ({ allow, updatedInput }) => {
          if (allow) resolve({ behavior: 'allow', updatedInput: updatedInput ?? input })
          else resolve({ behavior: 'deny', message: 'denied by host' })
        })
        out({ type: 'permission_request', requestId, toolName, input })
      })
  }
  return options
}

async function runChat(query, req) {
  const options = buildOptions(req)
  let capturedSessionId = req.sessionId ?? null
  let streamedText = false
  let streamedThinking = false
  let emittedTextChars = 0

  const q = query({ prompt: req.prompt, options })
  currentQuery = q
  try {
    for await (const event of q) {
      const e = event
      if (e.session_id && !capturedSessionId) {
        capturedSessionId = e.session_id
        out({ type: 'session_id', sessionId: capturedSessionId })
      }

      if (e.type === 'stream_event') {
        const se = e.event
        if (se?.type === 'content_block_delta') {
          const d = se.delta
          if (d?.type === 'text_delta' && typeof d.text === 'string') {
            streamedText = true
            emittedTextChars += d.text.length
            out({ type: 'text', text: d.text })
          } else if (d?.type === 'thinking_delta' && typeof d.thinking === 'string') {
            streamedThinking = true
            out({ type: 'thinking', text: d.thinking })
          }
        }
        continue
      }

      if (e.type === 'assistant') {
        const content = e.message?.content ?? []
        for (const b of content) {
          if (b.type === 'text' && typeof b.text === 'string' && !streamedText) {
            emittedTextChars += b.text.length
            out({ type: 'text', text: b.text })
          }
          if (b.type === 'thinking' && typeof b.thinking === 'string' && !streamedThinking) {
            out({ type: 'thinking', text: b.thinking })
          }
          if (b.type === 'tool_use') {
            out({ type: 'tool_call', name: b.name, input: b.input })
          }
        }
        streamedText = false
        streamedThinking = false
      }

      if (e.type === 'user') {
        const content = e.message?.content ?? []
        for (const b of content) {
          if (b.type === 'tool_result') out({ type: 'tool_result', content: b.content })
        }
      }

      if (e.type === 'result') {
        if (emittedTextChars === 0 && typeof e.result === 'string' && e.result.trim()) {
          out({ type: 'text', text: e.result })
        }
        out({ type: 'done', sessionId: capturedSessionId })
        currentQuery = null
        return
      }
    }
    // stream ended without an explicit result
    out({ type: 'done', sessionId: capturedSessionId })
  } catch (err) {
    out({ type: 'error', message: String(err?.message ?? err) })
  } finally {
    currentQuery = null
  }
}

async function main() {
  const sdk = await import('@anthropic-ai/claude-agent-sdk').catch(() => null)
  if (!sdk?.query) {
    out({ type: 'error', message: '@anthropic-ai/claude-agent-sdk not installed (run npm install in node-executor/)' })
    process.exit(1)
  }
  out({ type: 'ready' })

  const rl = createInterface({ input: process.stdin })
  for await (const line of rl) {
    const trimmed = line.trim()
    if (!trimmed) continue
    let req
    try {
      req = JSON.parse(trimmed)
    } catch {
      out({ type: 'error', message: 'invalid json request' })
      continue
    }
    if (req.type === 'chat') {
      // fire-and-forget: do NOT await, so stdin stays free to receive
      // permission_response / abort while the turn runs.
      runChat(sdk.query, req).catch((e) => out({ type: 'error', message: String(e?.message ?? e) }))
    } else if (req.type === 'permission_response') {
      const resolve = pendingPermissions.get(req.requestId)
      if (resolve) {
        pendingPermissions.delete(req.requestId)
        resolve({ allow: !!req.allow, updatedInput: req.updatedInput })
      }
    } else if (req.type === 'abort') {
      if (currentQuery?.interrupt) currentQuery.interrupt().catch(() => {})
    }
  }
}

main().catch((e) => {
  out({ type: 'error', message: String(e?.message ?? e) })
  process.exit(1)
})

#!/usr/bin/env node
// Spawns executor.mjs, sends one chat request, asserts we get session_id + text + done.
// Skips (exit 0) if `claude` is not on PATH.
import { spawn, execSync } from 'node:child_process'
import { createInterface } from 'node:readline'
import { fileURLToPath } from 'node:url'
import { dirname, join } from 'node:path'

const here = dirname(fileURLToPath(import.meta.url))

try {
  execSync('command -v claude', { stdio: 'ignore' })
} catch {
  console.log('SKIP: claude not on PATH')
  process.exit(0)
}

const child = spawn('node', [join(here, 'executor.mjs')], { stdio: ['pipe', 'pipe', 'inherit'] })
const rl = createInterface({ input: child.stdout })

let sawSession = false
let sawText = false
let sawDone = false
const timer = setTimeout(() => { console.error('TIMEOUT'); child.kill(); process.exit(1) }, 90_000)

rl.on('line', (line) => {
  let ev
  try { ev = JSON.parse(line) } catch { return }
  if (ev.type === 'ready') {
    child.stdin.write(JSON.stringify({ type: 'chat', prompt: 'Reply with exactly the word pong', cwd: '/tmp' }) + '\n')
  } else if (ev.type === 'session_id') sawSession = true
  else if (ev.type === 'text') sawText = true
  else if (ev.type === 'error') { console.error('ERROR event:', ev.message); clearTimeout(timer); child.kill(); process.exit(1) }
  else if (ev.type === 'done') {
    sawDone = true
    clearTimeout(timer)
    child.stdin.end()
    if (sawSession && sawText && sawDone) { console.log('OK: session_id + text + done'); process.exit(0) }
    console.error(`FAIL: session=${sawSession} text=${sawText} done=${sawDone}`)
    process.exit(1)
  }
})

child.on('exit', (code) => { if (!sawDone) { console.error('executor exited early, code', code); process.exit(1) } })

Self-Review ​

  • 스펙 커버리지: server.ts handleClaudeMessage 의 session_id/stream_event(text·thinking delta)/assistant(tool_use, 미스트림 시 text·thinking)/user(tool_result)/result(잔여 text·done) + canUseTool → 전부 executor.mjs 로 이식. settingSources 파리티, includePartialMessages 스트리밍, resume, extendedThinking 포함.
